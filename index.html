Google Apps Script Web App 串接範本

將下方程式碼加入你的 Apps Script 專案並部署為網頁應用程式，即可讓此頁面讀取你的試算表數據： 

function doGet() {
  const sh = SpreadsheetApp.getActiveSheet();
  const rows = sh.getDataRange().getValues();
  const data = rows.slice(1).filter(r => r[0]).map(r => ({
    code: String(r[0]),
    name: r[1],
    close: Number(r[2]),
    open: Number(r[3]),
    high: Number(r[4]),
    low: Number(r[5]),
    volume: Number(r[6]),
    date: String(r[8]),
    rsi14: Number(r[9]),
    signal: String(r[10]),
    conf: Number(r[11])
  }));
  return ContentService
    .createTextOutput(JSON.stringify({ ok: true, data }))
    .setMimeType(ContentService.MimeType.JSON);
}
